        // --- HIỆU ỨNG LÁ RƠI THIỀN ĐỊNH ---

        function createLeaf() {

            const leaf = document.createElement('div');

            leaf.classList.add('leaf');

            leaf.style.left = Math.random() * 100 + 'vw';

            const duration = Math.random() * 8 + 7;

            leaf.style.animationDuration = duration + 's';

            leaf.style.transform = `scale(${Math.random() * 0.5 + 0.5})`;

            document.body.appendChild(leaf);

            setTimeout(() => { leaf.remove(); }, duration * 1000);

        }

        setInterval(createLeaf, 800);


        // --- HIỆU ỨNG PHÁO HOA CHÚC MỪNG ---

        function shootFireworks() {

            var duration = 4000; // Pháo hoa nổ trong 4 giây

            var animationEnd = Date.now() + duration;

            var defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 100 };


            function randomInRange(min, max) { return Math.random() * (max - min) + min; }


            var interval = setInterval(function() {

                var timeLeft = animationEnd - Date.now();

                if (timeLeft <= 0) { return clearInterval(interval); }


                var particleCount = 40 * (timeLeft / duration);

                

                // Nổ pháo hoa từ 2 bên màn hình (màu vàng đồng và xanh lá)

                confetti(Object.assign({}, defaults, { 

                    particleCount, 

                    origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 },

                    colors: ['#D4AF37', '#e8f3e2', '#a8e6cf']

                }));

                confetti(Object.assign({}, defaults, { 

                    particleCount, 

                    origin: { x: randomInRange(0.7, 0.9), y: Math.random() - 0.2 },

                    colors: ['#D4AF37', '#e8f3e2', '#a8e6cf']

                }));

            }, 250);

        }


        // --- HÀNH ĐỘNG KHI BẤM NÚT "CHẠM MỞ BẤT NGỜ" ---

        function openInvitation() {

            // 1. Phát nhạc

            const music = document.getElementById('bgMusic');

            music.volume = 0.5; // Đặt âm lượng 50%

            music.play().catch(e => console.log("Trình duyệt chặn autoplay âm thanh."));


            // 2. Ẩn thiệp chúc mừng cũ

            const card1 = document.getElementById('greeting-card');

            const card2 = document.getElementById('invitation-card');

            

            card1.classList.add('fade-out');


            // 3. Hiện thiệp mời và bắn pháo hoa

            setTimeout(() => {

                card1.style.display = 'none'; // Giấu hẳn card 1

                card2.classList.remove('hidden-start');

                card2.classList.add('fade-in');

                

                // Nền hơi mờ đi một chút để tập trung vào thiệp mời

                document.getElementById('bg-img').style.filter = 'blur(4px) brightness(0.8)';

                

                // Bắn pháo hoa 

                shootFireworks();


            }, 800); // Đợi 0.8s cho thiệp 1 mờ dần

        }
